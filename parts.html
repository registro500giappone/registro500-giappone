async function debugSearch() { const query = document.getElementById('search-input').value.trim(); const resultsDiv = document.getElementById('results'); if (!query) { alert("キーワードを入れてください"); return; } resultsDiv.innerHTML = "DBから生データを取得中..."; // 検索対象を大幅に広げ、URLやID、品番のどこかに含まれていれば出すように変更 const { data, error } = await _supabase .from('parts') .select('*') .or(`name_jp.ilike.%${query}%,name_en.ilike.%${query}%,oem_no.ilike.%${query}%,page_url.ilike.%${query}%,product_no.ilike.%${query}%`) .limit(100); if (error) { resultsDiv.innerHTML = "エラー発生: " + error.message; return; } resultsDiv.innerHTML = ""; if (!data || data.length === 0) { resultsDiv.innerHTML = `DB内に "${query}" に該当するデータはありません。別の言葉（例: ヘッドライト）で試してください。`; return; } data.forEach(part => { const card = document.createElement('div'); card.className = 'debug-card'; card.innerHTML = `
ID: ${part.id} SHOP: ${part.shop_name}
OEM/品番: ${part.oem_no || 'N/A'}
商品番号: ${part.product_no || 'N/A'}
日本語名: ${part.name_jp}
英語名: ${part.name_en}
適合車種: ${part.target_cars}
DB保存価格: €${part.price_euro}
実際のショップページを開く `; resultsDiv.appendChild(card); }); }
